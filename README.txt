SQL پیشنهادی برای جدول posts:
create table if not exists public.posts (
 id bigint generated by default as identity primary key,
 title text not null, content text not null,
 kind text not null default 'article' check(kind in ('article','news')),
 published boolean not null default false, created_at timestamptz default now()
);
alter table public.posts enable row level security;

-- اجازه خواندن مطالب منتشرشده برای بازدیدکنندگان
create policy "Public reads published posts" on public.posts for select to anon, authenticated using (published = true);

-- پنل مدیریت را به UID دقیق حساب ادمین محدود کنید. UUID ادمین را جایگزین کنید:
create policy "Admin full access" on public.posts for all to authenticated
using (auth.uid() = 'ADMIN-UUID'::uuid)
with check (auth.uid() = 'ADMIN-UUID'::uuid);

grant select on public.posts to anon, authenticated;
grant insert, update, delete on public.posts to authenticated;
grant usage, select on sequence public.posts_id_seq to authenticated;

از Supabase Auth، ثبت‌نام عمومی (Signups) را غیرفعال کنید. UUID مدیر از Authentication > Users قابل مشاهده است.

جایگاه‌های تبلیغات در سایت فعلاً کادرهای رزرو هستند و تبلیغ واقعی باید با تصویر/لینک یا سامانه تبلیغات جایگزین شود.


SQL آمار بازدید واقعی:
این SQL را در Supabase > SQL Editor اجرا کنید. هر مطلب برای هر نشست مرورگر یک بار بازدید می‌شود.
برای جلوگیری از دستکاری آمار، در صورت نیاز می‌توانید CAPTCHA یا محدودسازی سمت سرور اضافه کنید.

create table if not exists public.post_views (
  id bigint generated by default as identity primary key,
  post_id bigint not null references public.posts(id) on delete cascade,
  viewed_at timestamptz not null default now()
);

alter table public.post_views enable row level security;

create or replace function public.record_post_view(p_post_id bigint)
returns void
language plpgsql
security definer
set search_path = public
as $$
begin
  if exists (select 1 from public.posts where id = p_post_id and published = true) then
    insert into public.post_views(post_id) values (p_post_id);
  end if;
end;
$$;

create or replace function public.get_post_views(p_post_id bigint)
returns bigint
language sql
security definer
set search_path = public
as $$
  select count(*) from public.post_views where post_id = p_post_id;
$$;

create or replace function public.get_site_stats()
returns table(total_views bigint, today_views bigint)
language sql
security definer
set search_path = public
as $$
  select
    count(*)::bigint,
    count(*) filter (where viewed_at >= date_trunc('day', now()))::bigint
  from public.post_views;
$$;

revoke all on function public.record_post_view(bigint) from public;
revoke all on function public.get_post_views(bigint) from public;
revoke all on function public.get_site_stats() from public;
grant execute on function public.record_post_view(bigint) to anon, authenticated;
grant execute on function public.get_post_views(bigint) to anon, authenticated;
grant execute on function public.get_site_stats() to anon, authenticated;

=== جستجو و اصلاح نمایش اخبار ===
در این نسخه پنل ادمین جستجو بر اساس عنوان و متن دارد و می‌توان فهرست را به خبر یا مقاله محدود کرد.
در سایت، خبرهای منتشرشده با kind=news نمایش داده می‌شوند؛ خبرهای آرشیوشده از فهرست عمومی اخبار مخفی می‌مانند.
برای نمایش یک خبر، در پنل نوع مطلب را روی «خبر» و وضعیت انتشار را روی «منتشر شود» قرار دهید.
\n\nموضوع تبلیغ (topic): users, companies, shared_resources, livestock, poultry, feed, equipment, services, products, chicks, other.\nمحل نمایش (placement): all, home, sidebar, news, articles, livestock, poultry, feed.\n
